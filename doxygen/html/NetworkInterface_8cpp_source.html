84 if( family != AF_INET && family != AF_INET6)
85 continue;

88 std::string broadcast;
89 std::string destination;

92 // getifaddrs implementations, so the flags determine which is valid.
93 if( (entry->ifa_flags & IFF_BROADCAST) != 0)
94 broadcast = addressString(entry->ifa_broadaddr);

97 destination = addressString(entry->ifa_dstaddr);

99 ret.push_back( NetworkAddressInfo{
