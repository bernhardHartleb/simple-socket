84int InternetSocket::timedConnect( std::string_view foreignAddress, unsigned short foreignPort, int timeout)
85{
86 SockAddress addr( m_family, foreignAddress, foreignPort);
87
88 // Set O_NONBLOCK and restore blocking state on destruction
89 NonBlockingGuard nonBlocking( m_socket);
90
91 const auto throwError = []() {
92 throw SocketException("timedConnect failed (connect)");
93 };
94
95 // Asynchronous connection attempt
96 int ret = ::connect( m_socket, addr.data(), addr.length());
97 if( ret == 0) { // Already connected
98 m_peerDisconnected = false;
99 return 1;
100 }
101
102 // Did connect return an unexpected error?
103 if( errno != EINPROGRESS && errno != EWOULDBLOCK && errno != EINTR)
104 throwError();
105
106 // Wait for the connection to complete
107 struct pollfd poll;
108 poll.fd = m_socket;
109 poll.events = POLLOUT;
110
111 ret = TEMP_FAILURE_RETRY (::poll( &poll, 1, timeout));
112 if( ret < 0) throwError();
113 if( ret == 0) { // Timeout
114 errno = ETIMEDOUT;
115 return 0;
116 }
117
118 // Check for socket errors after poll success
119 int sockError = 0;
120 socklen_t errLen = sizeof(sockError);
121 ret = ::getsockopt( m_socket, SOL_SOCKET, SO_ERROR, &sockError, &errLen);
122 if( ret < 0) throwError();
123 if( sockError != 0) { // Socket error
124 errno = sockError;
125 throwError();
126 }
127
128 m_peerDisconnected = false;
129 return 1;
130}
