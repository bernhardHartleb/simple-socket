63int UDPSocket::timedReceiveFrom( void* buffer, size_t len, std::string& sourceAddress, unsigned short& sourcePort, int timeout)
64{
65 struct pollfd poll;
66 poll.fd = m_socket;
67 poll.events = POLLIN | POLLPRI | POLLRDHUP;
68
69 int ret = TEMP_FAILURE_RETRY (::poll( &poll, 1, timeout));
70
71 if( ret == 0) return 0;
72 if( ret < 0) throw SocketException("timedReceive failed (poll)");
