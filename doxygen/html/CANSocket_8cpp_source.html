71int CANSocket::getInterfaceIndex( std::string_view interface) const
72{
73 const size_t len = interface.length();
74 // binds to all interfaces
75 if( len == 0) return 0;
76
77 // needed space is size plus null character
78 if( len >= IF_NAMESIZE)
79 throw SocketException("Interface name is too long", false);
80
81 struct ifreq ifr;
82 std::memcpy( ifr.ifr_name, interface.data(), len);
83 ifr.ifr_name[len] = 0;
84
85 if( ioctl( m_socket, SIOCGIFINDEX, &ifr) < 0)
86 throw SocketException("ioctl failed (getInterfaceIndex)");
