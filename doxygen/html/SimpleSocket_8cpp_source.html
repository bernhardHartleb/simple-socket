232int SimpleSocket::timedAccept( int timeout) const
233{
234 // negative timeout means wait forever
235 if( timeout < 0) return accept();
236
237 bool firstRun = true;
238 using Clock = std::chrono::steady_clock;
239 auto now = Clock::now();
240 const auto deadline = now + std::chrono::milliseconds(timeout);
241
242 while(1)
243 {
244 // ensure now == deadline when timeout is zero
245 if( !firstRun) now = Clock::now();
246 // invalid socket handle on timeout
247 if( now >= deadline && !firstRun) return -1;
248 firstRun = false;
249
250 auto remaining = std::chrono::ceil<std::chrono::milliseconds>(deadline - now);
251 const int pollTimeout = static_cast<int>(remaining.count());
252
253 pollfd pfd {};
254 pfd.fd = m_socket;
255 pfd.events = POLLIN;
256
257 int ret = ::poll( &pfd, 1, pollTimeout);
258 if( ret == 0) return -1; // invalid socket handle on timeout
259 if( ret < 0)
260 {
261 if( errno == EINTR) continue;
262 throw SocketException("timedAccept failed (poll)");
263 }
264
265 if( pfd.revents & POLLNVAL)
266 throw SocketException("timedAccept failed (invalid socket)", false);
267
268 if( pfd.revents & POLLIN)
269 {
270 int fd = acceptNative();
271 if( fd >= 0) return fd;
272
273 const int error = errno;
274 if( error == EINTR
275 || error == EAGAIN || error == EWOULDBLOCK
276 || detail::isTransientAcceptError(error))
277 continue;
278
279 throw SocketException("timedAccept failed (accept)", error);
280 }
281
282 if( pfd.revents & POLLERR)
283 throw SocketException("timedAccept failed (POLLERR)", false);
284
285 if( pfd.revents & POLLHUP)
286 throw SocketException("timedAccept failed (listening socket hung up)", false);
287 }
288}
