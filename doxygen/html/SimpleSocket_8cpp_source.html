52 int flags = 0;

89int SimpleSocket::timedReceive( void* buffer, size_t len, int timeout)
90{
91 struct pollfd poll;
92 poll.fd = m_socket;
93 poll.events = POLLIN | POLLPRI | POLLRDHUP;
94
95 int ret = TEMP_FAILURE_RETRY (::poll( &poll, 1, timeout));
96
97 if( ret == 0) return 0;
98 if( ret < 0) throw SocketException("timedReceive failed (poll)");
99
100 if( poll.revents & POLLRDHUP || poll.revents & POLLHUP)
101 m_peerDisconnected = true;
102
103 if( poll.revents & POLLIN || poll.revents & POLLPRI)
104 return receive( buffer, len);
105
106 if( poll.revents & (POLLERR | POLLNVAL))
107 throw SocketException("timedReceive failed (poll)");
108
109 return 0;
110}
